Data Simulation/Acquisition Functions --- @st.cache_data def load_and_simulate_data(crop, region): """ Simulates historical price and weather data based on crop/region characteristics. In a real project, this would be API calls to CEPEA, IBGE, etc. """ today = date.today() start_date = today - timedelta(days=5*365) # 5 years of data # Base parameters inspired by real CEPEA/IBGE trends (e.g., higher soy price volatility) if crop == "Soja (Soybean)": base_price = 150.00 volatility = 10.0 yearly_growth = 0.05 elif crop == "Café Arábica (Arabica Coffee)": base_price = 800.00 volatility = 50.0 yearly_growth = 0.03 else: # Milho (Corn) base_price = 60.00 volatility = 5.0 yearly_growth = 0.04 dates = pd.to_datetime(pd.date_range(start=start_date, end=today, freq='D')) n_days = len(dates) # Price series with trend, noise, and growth prices = base_price * (1 + yearly_growth * (dates.year - dates.year.min())) + \ np.random.normal(0, volatility, n_days) + \ volatility * np.sin(dates.dayofyear / 365 * 2 * np.pi) # Simple seasonality prices = np.maximum(prices, base_price * 0.8) # Prevent extreme drops df = pd.DataFrame({'ds': dates, 'y': prices.round(2)}) # Simulate weather as a REGRESSOR (for advanced forecasting) # Average rainfall (mm) - a major factor for yield/sale volume avg_rainfall = 5 + 15 * np.sin(dates.dayofyear / 365 * 2 * np.pi) + np.random.normal(0, 3, n_days) df['rainfall_mm'] = np.maximum(avg_rainfall, 0).round(2) return df @st.cache_data def get_current_weather(region): """Simulates real-time weather API call for a selected region.""" # In a real app, you would use: # lat, lon = WEATHER_LAT_LON.get(region) # response = requests.get(f"http://api.openweathermap.org/data/2.5/weather?lat={lat}&lon={lon}&appid={WEATHER_API_KEY}&units=metric&lang=pt") # return response.json() # Mock data for demonstration temp = np.random.randint(20, 35) precip = np.random.choice([0.0, 2.5, 10.0]) condition = np.random.choice(["Céu Limpo", "Nublado", "Chuva Forte"]) return { "Temperatura Atual": f"{temp}°C", "Precipitação (24h)": f"{precip} mm", "Condição": condition } # --- 2. Prophet Forecasting Model --- @st.cache_data(show_spinner=False) def run_prophet_forecast(df, periods=365): """Runs the Prophet model to forecast prices.""" m = Prophet( seasonality_mode='multiplicative', changepoint_prior_scale=0.05, # Adjust for trend volatility yearly_seasonality=True, weekly_seasonality=False, daily_seasonality=False ) # Add external regressor (simulated weather data) m.add_regressor('rainfall_mm') m.fit(df) # Create future dataframe for forecasting future = m.make_future_dataframe(periods=periods) # Simulate future weather (e.g., based on average historical patterns) future_dates = pd.to_datetime(future['ds']) future['rainfall_mm'] = (5 + 15 * np.sin(future_dates.dayofyear / 365 * 2 * np.pi) + np.random.normal(0, 3, len(future))).clip(lower=0).round(2) forecast = m.predict(future) return m, forecast # --- 3.
